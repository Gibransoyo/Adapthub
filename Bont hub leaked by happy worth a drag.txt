local function mkCorner(parent, radius)
    local corner = Instance.new("UICorner", parent)
    corner.CornerRadius = UDim.new(0, radius or 6)
    return corner
end

local function mkStroke(parent, color, thickness, transparency)
    local stroke = Instance.new("UIStroke", parent)
    stroke.Color = color or C.border
    stroke.Thickness = thickness or 1
    stroke.Transparency = transparency or 0
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    return stroke
end

local function mkPadding(parent, left, top, right, bottom)
    local padding = Instance.new("UIPadding", parent)
    padding.PaddingLeft = UDim.new(0, left or 0)
    padding.PaddingTop = UDim.new(0, top or 0)
    padding.PaddingRight = UDim.new(0, right or 0)
    padding.PaddingBottom = UDim.new(0, bottom or 0)
    return padding
end

local function makeDraggable(frame, handle)
    local src = handle or frame
    local dragging = false
    local dragInput = nil
    local dragStart = nil
    local startPos = nil
 
    src.InputBegan:Connect(function(input)
        if State.uiLocked then return end
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = frame.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
 
    src.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
 
    UIS.InputChanged:Connect(function(input)
        if input == dragInput and dragging and not State.uiLocked then
            local delta_x = input.Position.X - dragStart.X
            local delta_y = input.Position.Y - dragStart.Y
            frame.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta_x,
                startPos.Y.Scale, startPos.Y.Offset + delta_y
            )
        end
    end)
end

local function cleanupOldGuis()
    local names = {"BontHubGui", "BontHubV26", "BontHeadTag", "RagdollTimerGUI", "AnimationChanger", "BontHubMini"}
    for _, name in pairs(names) do
        pcall(function()
            local o = CoreGui:FindFirstChild(name)
            if o then o:Destroy() end
        end)
        pcall(function()
            local o = LP:WaitForChild("PlayerGui"):FindFirstChild(name)
            if o then o:Destroy() end
        end)
        pcall(function()
            local hidden = CoreGui:FindFirstChild("HiddenUI")
            if hidden then
                local o = hidden:FindFirstChild(name)
                if o then o:Destroy() end
            end
        end)
    end
end

                lconn = nil

            if buttonName then
                kbtn.Text = getControllerDisplayName(buttonName)
                kbtn.TextColor3 = C.pink
            else
                kbtn.Text = "NONE"
                kbtn.TextColor3 = C.textDim

local GRID_W = COLS * (BTN_W + BTN_GAP) - BTN_GAP
local GRID_H = math.ceil(#stackDefs / COLS) * (BTN_H + BTN_GAP) - BTN_GAP

local function getDefaultStackPos(i)
    local col = (i - 1) % COLS
    local row = math.floor((i - 1) / COLS)
    return UDim2.new(
        1, -(GRID_W + 14) + col * (BTN_W + BTN_GAP),
        0.5, -(GRID_H / 2) + row * (BTN_H + BTN_GAP)
    )
end

                doTpDown()

                instantReset()
                return
            end
            if def.key == "carrySpeed" then
                State.speedToggled = not State.speedToggled
                setOn(State.speedToggled)
                return
            end
     
            local newState = not btnState
            setOn(newState)
     
            if def.key == "autoLeft" then
                State.autoLeftEnabled = newState
                if newState then
                    if State.batAimbotToggled then
                        State.batAimbotToggled = false
                        stopBatAimbot()
                        if stackBtnRefs.aimbot then
                            stackBtnRefs.aimbot.setOn(false)
                        end
                    end
                    if State.batV2Toggled then
                        State.batV2Toggled = false
                        stopBatV2()
                        if stackBtnRefs.batV2 then
                            stackBtnRefs.batV2.setOn(false)
                        end
                    end
                    startAutoLeft()
                else
                    stopAutoLeft()
                end
            elseif def.key == "autoRight" then
                State.autoRightEnabled = newState
                if newState then
                    if State.batAimbotToggled then
                        State.batAimbotToggled = false
                        stopBatAimbot()
                        if stackBtnRefs.aimbot then
                            stackBtnRefs.aimbot.setOn(false)
                        end
                    end
                    if State.batV2Toggled then
                        State.batV2Toggled = false
                        stopBatV2()
                        if stackBtnRefs.batV2 then
                            stackBtnRefs.batV2.setOn(false)
                        end
                    end
                    startAutoRight()
                else
                    stopAutoRight()
                end
            elseif def.key == "aimbot" then
                State.batAimbotToggled = newState
                if newState then
                    if State.autoLeftEnabled then
                        State.autoLeftEnabled = false
                        stopAutoLeft()
                        if stackBtnRefs.autoLeft then
                            stackBtnRefs.autoLeft.setOn(false)
                        end
                    end
                    if State.autoRightEnabled then
                        State.autoRightEnabled = false
                        stopAutoRight()
                        if stackBtnRefs.autoRight then
                            stackBtnRefs.autoRight.setOn(false)
                        end
                    end
                    if State.batV2Toggled then
                        State.batV2Toggled = false
                        stopBatV2()
                        if stackBtnRefs.batV2 then
                            stackBtnRefs.batV2.setOn(false)
                        end
                    end
                    startBatAimbot()
                else
                    stopBatAimbot()
                end
            elseif def.key == "batV2" then
                State.batV2Toggled = newState
                if newState then
                    if State.autoLeftEnabled then
                        State.autoLeftEnabled = false
                        stopAutoLeft()
                        if stackBtnRefs.autoLeft then
                            stackBtnRefs.autoLeft.setOn(false)
                        end
                    end
                    if State.autoRightEnabled then
                        State.autoRightEnabled = false
                        stopAutoRight()
                        if stackBtnRefs.autoRight then
                            stackBtnRefs.autoRight.setOn(false)
                        end
                    end
                    if State.batAimbotToggled then
                        State.batAimbotToggled = false
                        stopBatAimbot()
                        if stackBtnRefs.aimbot then
                            stackBtnRefs.aimbot.setOn(false)
                        end
                    end
                    startBatV2()
                else
                    stopBatV2()
                end
            elseif def.key == "lagger" then
                State.laggerEnabled = newState
                if newState then
                    State._prevCarry = State.carrySpeed
                    State._prevSpeed = State.speedToggled
                    State.speedToggled = false

                        stackBtnRefs.carrySpeed.setOn(false)

                    State.speedToggled = State._prevSpeed or false

                    end
                end
            elseif def.key == "drop" then
                if newState then
                    runDropBrainrot()
                else
                    stopDropBrainrot()
                end
            end
        end
     
        makeDraggable(btnFrame, btnFrame)
     
        -- Tap to toggle
        local clickArea = Instance.new("TextButton", btnFrame)
        clickArea.Size = UDim2.new(1, 0, 1, 0)
        clickArea.BackgroundTransparency = 1
        clickArea.Text = ""
        clickArea.ZIndex = 9992

            local moved = false
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    if not moved and not State.uiLocked then
                        State.guiVisible = not State.guiVisible
                        if mainOuter then mainOuter.Visible = State.guiVisible end
                    end
                end
            end)
            UIS.InputChanged:Connect(function(changeInput)
                if changeInput.UserInputType == Enum.UserInputType.MouseMovement or changeInput.UserInputType == Enum.UserInputType.Touch then
                    if (changeInput.Position - startPos).Magnitude > 10 then
                        moved = true
                    end
                end
            end)
        end
    end)

    task.spawn(function()
        local t = 0
        while true do
            t = t + 0.03
            local alpha = (math.sin(t) + 1) / 2
            pcall(function()
                miniStroke.Transparency = alpha * 0.7
            end)
            task.wait(0.05)
        end
    end)

    return miniBtn

-- GAME LOGIC - Same as Bont Hub
-- ============================================================
local function initGameLogic()
    local function resetProgressBar()
        if stealBarPctLbl then stealBarPctLbl.Text = "0%" end
        if stealBarFill then stealBarFill.Size = UDim2.new(0, 0, 1, 0) end
        if statusDot then statusDot.BackgroundColor3 = C.barStatusOff end
    end

    doTpDown = function()
        pcall(function()
            local char = LP.Character
            if not char then return end
            local hrp = char:FindFirstChild("HumanoidRootPart")
            if not hrp then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.FloorMaterial ~= Enum.Material.Air then return end
            hrp.CFrame = CFrame.new(hrp.Position.X, -7, hrp.Position.Z) * CFrame.Angles(0, select(2, hrp.CFrame:ToEulerAnglesYXZ()), 0)
            hrp.AssemblyLinearVelocity = Vector3.zero
        end)
    end

    startAutoTpDown = function()
        if Conns.autoTpDown then return end
        Conns.autoTpDown = RunService.Heartbeat:Connect(function()
            if not State.autoTpDownEnabled then return end
            local char = LP.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum and hum.FloorMaterial ~= Enum.Material.Air then return end
            if root.Position.Y >= State.autoTpTargetY then
                root.CFrame = CFrame.new(root.Position.X, -7, root.Position.Z) * CFrame.Angles(0, select(2, root.CFrame:ToEulerAnglesYXZ()), 0)
                root.AssemblyLinearVelocity = Vector3.zero
            end
        end)
    end

    stopAutoTpDown = function()
        if Conns.autoTpDown then
            Conns.autoTpDown:Disconnect()
            Conns.autoTpDown = nil
        end
    end

    local MOVE_KEYS = {
        [Enum.KeyCode.W] = true, [Enum.KeyCode.A] = true,
        [Enum.KeyCode.S] = true, [Enum.KeyCode.D] = true,
        [Enum.KeyCode.Up] = true, [Enum.KeyCode.Left] = true,
        [Enum.KeyCode.Down] = true, [Enum.KeyCode.Right] = true,
    }

    local function getActiveMoveSpeed()
        return State.laggerEnabled and State.laggerSpeed or (State.speedToggled and State.carrySpeed or State.normalSpeed)
    end

    local function isRagdollState(hum)
        if not hum then return true end
        local st = hum:GetState()
        return hum.PlatformStand or
               st == Enum.HumanoidStateType.Physics or
               st == Enum.HumanoidStateType.Ragdoll or
               st == Enum.HumanoidStateType.FallingDown
    end

    local function isAimbotActive()
        return State.batAimbotToggled or State.batV2Toggled
    end

    -- Speed loop
    RunService.RenderStepped:Connect(function()
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local hrp = char:FindFirstChild("HumanoidRootPart")
        if not hum or not hrp then return end

        if isRagdollState(hum) then
            State.lastMoveDir = Vector3.new(0, 0, 0)
            return
        end

        if isAimbotActive() then
            hum.JumpPower = 0
        else
            hum.JumpPower = 60
        end

        if not State.batAimbotToggled and not State.batV2Toggled and not State.autoLeftEnabled and not State.autoRightEnabled then
            local md = hum.MoveDirection
            local spd = getActiveMoveSpeed()
            if md.Magnitude > 0 then
                State.lastMoveDir = md
                hrp.Velocity = Vector3.new(md.X * spd, hrp.Velocity.Y, md.Z * spd)
            elseif State.antiRagdollEnabled and State.lastMoveDir.Magnitude > 0 then
                local anyHeld = false
                for key in pairs(MOVE_KEYS) do
                    if UIS:IsKeyDown(key) then
                        anyHeld = true
                        break
                    end
                end
                if anyHeld then
                    hrp.Velocity = Vector3.new(State.lastMoveDir.X * spd, hrp.Velocity.Y, State.lastMoveDir.Z * spd)
                end
            end
        end

        if State.flingProtection then
            local vel = hrp.AssemblyLinearVelocity
            if vel.Magnitude > 200 then
                hrp.AssemblyLinearVelocity = vel * 0.3
                hrp.AssemblyAngularVelocity = Vector3.zero
            end
        end

        if speedLabel then
            speedLabel.Text = string.format("Speed: %.1f", Vector3.new(hrp.Velocity.X, 0, hrp.Velocity.Z).Magnitude)
        end
    end)

    local function canMoveAuto()
        local char = LP.Character
        if not char then return false end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if not hum or not root then return false end
        if root.Anchored then return false end
        local st = hum:GetState()
        if st == Enum.HumanoidStateType.Physics then return false end
        if st == Enum.HumanoidStateType.Ragdoll then return false end

    local function stopAutoMove()
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        local root = char:FindFirstChild("HumanoidRootPart")
        if hum then hum:Move(Vector3.zero, false) end
        if root then
            local v = root.Velocity
            root.Velocity = Vector3.new(v.X * 0.1, v.Y, v.Z * 0.1)
        end
    end

    local function moveToAuto(target)
        local char = LP.Character
        if not char then return true end
        local root = char:FindFirstChild("HumanoidRootPart")
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not root or not hum then return true end
        local d = target - root.Position
        local flat = Vector3.new(d.X, 0, d.Z)
        if flat.Magnitude < 1.5 then return true end
        local dir = flat.Unit
        hum:Move(dir, false)
        local autoSpeed = State.normalSpeed
        root.Velocity = Vector3.new(dir.X * autoSpeed, root.Velocity.Y, dir.Z * autoSpeed)
        return false
    end

    -- ANTI RAGDOLL
    -- ============================================================
    startAntiRagdoll = function()
        if Conns.antiRag then Conns.antiRag:Disconnect(); Conns.antiRag = nil end
        local cachedChar, cachedHum, cachedRoot = nil, nil, nil

        local function refreshCache()
            local char = LP.Character
            if not char then return false end
            local hum = char:FindFirstChildOfClass("Humanoid")
            local root = char:FindFirstChild("HumanoidRootPart")
            if not hum or not root then return false end
            cachedChar = char
            cachedHum = hum
            cachedRoot = root
            return true
        end

        local function isRagdolled()
            if not cachedHum or not cachedHum.Parent then return true end
            local state = cachedHum:GetState()
            if state == Enum.HumanoidStateType.Physics or
               state == Enum.HumanoidStateType.Ragdoll or
               state == Enum.HumanoidStateType.FallingDown then
                return true
            end
            if cachedHum.PlatformStand then return true end
            local endTime = LP:GetAttribute("RagdollEndTime")
            if endTime then
                local now = workspace:GetServerTimeNow()
                if (endTime - now) > 0 then return true end
            end
            return false
        end

        local function forceExitRagdoll()
            if not cachedHum or not cachedRoot then return end
            pcall(function()
                LP:SetAttribute("RagdollEndTime", workspace:GetServerTimeNow())
            end)
            if cachedHum.Health > 0 then
                pcall(function()
                    cachedHum:ChangeState(Enum.HumanoidStateType.Running)
                end)
                task.defer(function()
                    if cachedHum and cachedHum.Parent and cachedHum.Health > 0 then
                        local st = cachedHum:GetState()
                        if st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll then
                            pcall(function()
                                cachedHum:ChangeState(Enum.HumanoidStateType.GettingUp)
                            end)
                            task.wait(0.05)
                            pcall(function()
                                cachedHum:ChangeState(Enum.HumanoidStateType.Running)
                            end)
                        end
                    end
                end)
            end
            cachedRoot.Anchored = false
            cachedRoot.AssemblyLinearVelocity = Vector3.zero
            cachedRoot.AssemblyAngularVelocity = Vector3.zero
            cachedRoot.CFrame = CFrame.new(cachedRoot.Position) * CFrame.Angles(0, math.rad(cachedRoot.Orientation.Y), 0)
            pcall(function()
                if Workspace.CurrentCamera.CameraSubject ~= cachedHum then
                    Workspace.CurrentCamera.CameraSubject = cachedHum
                end
            end)
        end

        refreshCache()

        Conns.antiRag = RunService.RenderStepped:Connect(function()
            if not cachedHum or not cachedHum.Parent or not cachedRoot or not cachedRoot.Parent then
                if not refreshCache() then return end
            end
            pcall(function()
                if Workspace.CurrentCamera.CameraSubject ~= cachedHum then
                    Workspace.CurrentCamera.CameraSubject = cachedHum
                end
            end)
            if isRagdolled() then
                for _, v in ipairs(cachedChar:GetDescendants()) do
                    if v:IsA("BallSocketConstraint") or (v:IsA("Attachment") and v.Name:find("Ragdoll")) then
                        pcall(function() v:Destroy() end)
                    end
                end
                forceExitRagdoll()
            end
        end)
    end

    stopAntiRagdoll = function()
        if Conns.antiRag then
            Conns.antiRag:Disconnect()
            Conns.antiRag = nil
        end
    end

    local function useMedusaCounter()
        if State.medusaDebounce then return end
        if tick() - State.medusaLastUsed < MEDUSA_COOLDOWN then return end
        local c = LP.Character
        if not c then return end
        State.medusaDebounce = true

        if not med then
            State.medusaDebounce = false
            return
        end
        if med.Parent ~= c then
            local h = c:FindFirstChildOfClass("Humanoid")
            if h then h:EquipTool(med) end
        end
        pcall(function() med:Activate() end)
        State.medusaLastUsed = tick()
        State.medusaDebounce = false
    end

    -- ============================================================
    -- BAT AIMBOT
    -- ============================================================
    local AUTO_BAT_SPEED = 58
    local AUTO_BAT_VERT_SPEED = 52
    local AUTO_BAT_DIST = -2.8
    local AUTO_BAT_HEIGHT = 4.75
    local AUTO_BAT_TURN_SPEED = 285
    local AUTO_BAT_MAX_TURN_RATE = 28
    local _autoBatLastScan = 0
    local _autoBatEquipped = false
    _aimbotTarget = nil

    local function _getBatAimbotTarget()
        local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if not root then return nil end
        
        local now = tick()
        if now - _autoBatLastScan <= 0.1 and _aimbotTarget and _aimbotTarget.Parent then
            local hum = _aimbotTarget.Parent:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health > 0 then return _aimbotTarget end
        end
        
        _autoBatLastScan = now
        _aimbotTarget = nil
        
        local closest, minDist = nil, math.huge
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if tRoot and hum and hum.Health > 0 then
                    local dist = (tRoot.Position - root.Position).Magnitude
                    if dist < minDist then
                        minDist = dist
                        closest = tRoot
                    end
                end
            end
        end
        
        _aimbotTarget = closest
        return _aimbotTarget
    end

    startBatAimbot = function()
        if Conns.aimbot then Conns.aimbot:Disconnect() end
        if State.autoLeftEnabled then
            State.autoLeftEnabled = false
            if stackBtnRefs.autoLeft then stackBtnRefs.autoLeft.setOn(false) end
            stopAutoLeft()
        end
        if State.autoRightEnabled then
            State.autoRightEnabled = false
            if stackBtnRefs.autoRight then stackBtnRefs.autoRight.setOn(false) end
            stopAutoRight()
        end
        if State.batV2Toggled then
            State.batV2Toggled = false
            if stackBtnRefs.batV2 then stackBtnRefs.batV2.setOn(false) end
            stopBatV2()
        end
        
        _autoBatEquipped = false
        local hum0 = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if hum0 then hum0.AutoRotate = false end

        Conns.aimbot = RunService.Heartbeat:Connect(function()
            if not State.batAimbotToggled then return end
            
            local char = LP.Character
            local hum = char and char:FindFirstChildOfClass("Humanoid")
            local root = char and char:FindFirstChild("HumanoidRootPart")
            if not root or not hum then return end
            
            if not _autoBatEquipped then
                _autoBatEquipped = true
                if not char:FindFirstChildOfClass("Tool") then
                    local bp = LP:FindFirstChildOfClass("Backpack") or LP:FindFirstChild("Backpack")
                    local bpBat = bp and bp:FindFirstChild("Bat")
                    if bpBat then pcall(function() hum:EquipTool(bpBat) end) end
                end
            end
            
            hum.JumpPower = 0
            local target = _getBatAimbotTarget()
            
            if target then
                local targetVel = target.AssemblyLinearVelocity
                local aimTargetPos = target.Position + (targetVel * math.clamp(targetVel.Magnitude / 130, 0.05, 0.15))
                hum.AutoRotate = false
                
                local look = aimTargetPos - root.Position
                local flatLook = Vector3.new(look.X, 0, look.Z)
                
                if look.Magnitude > 0.01 and flatLook.Magnitude > 0.01 then
                    local targetYaw = math.deg(math.atan2(-flatLook.X, -flatLook.Z))
                    local yawDelta = (targetYaw - root.Orientation.Y + 180) % 360 - 180
                    local targetPitch = math.deg(math.atan2(look.Y, flatLook.Magnitude))
                    local pitchDelta = (targetPitch - root.Orientation.X + 180) % 360 - 180
                    
                    local yawRate = math.clamp(math.rad(yawDelta) * AUTO_BAT_TURN_SPEED, -AUTO_BAT_MAX_TURN_RATE, AUTO_BAT_MAX_TURN_RATE)
                    local pitchRate = math.clamp(math.rad(pitchDelta) * AUTO_BAT_TURN_SPEED, -AUTO_BAT_MAX_TURN_RATE, AUTO_BAT_MAX_TURN_RATE)
                    
                    local yawRad = math.rad(root.Orientation.Y)
                    local rightAxis = Vector3.new(math.cos(yawRad), 0, -math.sin(yawRad))
                    root.AssemblyAngularVelocity = Vector3.new(0, yawRate, 0) + (rightAxis * pitchRate)
                else
                    root.AssemblyAngularVelocity = Vector3.zero
                end
                
                local dir = look.Magnitude > 0.01 and look.Unit or Vector3.zero
                local standPos = aimTargetPos - (dir * AUTO_BAT_DIST) + Vector3.new(0, AUTO_BAT_HEIGHT, 0)
                local moveDir = standPos - root.Position
                local hDir = Vector3.new(moveDir.X, 0, moveDir.Z)
                
                local hVel = hDir.Magnitude > 0.1 and hDir.Unit * AUTO_BAT_SPEED or Vector3.zero
                local vVel = math.abs(moveDir.Y) > 0.1 and Vector3.new(0, math.sign(moveDir.Y) * AUTO_BAT_VERT_SPEED, 0) or Vector3.new(0, -2, 0)
                
                root.AssemblyLinearVelocity = hVel + vVel
                if hDir.Magnitude > 0.5 then hum:Move(hDir.Unit, false) end
            else
                hum.AutoRotate = true
                root.AssemblyAngularVelocity = Vector3.zero
            end
            
            if State.autoSwingEnabled then
                local bat = char:FindFirstChild("Bat")
                if bat and bat:IsA("Tool") then
                    pcall(function() bat:Activate() end)
                end
            end
        end)
    end

    stopBatAimbot = function()

    local function _v2GetClosest()
        local root = LP.Character and LP.Character:FindFirstChild("HumanoidRootPart")
        if not root then return nil, math.huge end
        local closest, minDist = nil, math.huge
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                local tRoot = plr.Character:FindFirstChild("HumanoidRootPart")
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if tRoot and hum and hum.Health > 0 then
                    local dist = (tRoot.Position - root.Position).Magnitude
                    if dist < minDist then
                        minDist = dist
                        closest = tRoot
                    end
                end
            end
        end
        return closest, minDist
    end

    startBatV2 = function()
        if Conns.batV2 then Conns.batV2:Disconnect() end
        local hum0 = LP.Character and LP.Character:FindFirstChildOfClass("Humanoid")
        if hum0 then
            if _v2PrevAutoRotate == nil then _v2PrevAutoRotate = hum0.AutoRotate end
            hum0.AutoRotate = false
        end
 
        Conns.batV2 = RunService.RenderStepped:Connect(function()
            if not State.batV2Toggled then return end
            local char = LP.Character
            if not char then return end
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then return end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if not hum then return end

            hum.JumpPower = 0

            if not char:FindFirstChildOfClass("Tool") then
                local bat = _v2FindBat()
                if bat then pcall(function() hum:EquipTool(bat) end) end
            end

            local target, targetDist = _v2GetClosest()
            if not target then return end

            local myPos = root.Position
            local targetPos = target.Position
            local direction = targetPos - myPos
            local flatDir = Vector3.new(direction.X, 0, direction.Z)
            if flatDir.Magnitude > 0 then flatDir = flatDir.Unit else flatDir = Vector3.zero end

            local chaseSpeed = 59.6
            local desiredHeight = targetPos.Y + 3.5
            local yVel = (desiredHeight - myPos.Y) * 15
            if hum.FloorMaterial ~= Enum.Material.Air then yVel = math.max(yVel, 10) end
            yVel = math.clamp(yVel, -35, 40)

            local desiredVel = Vector3.new(flatDir.X * chaseSpeed, yVel, flatDir.Z * chaseSpeed)
            root.AssemblyLinearVelocity = root.AssemblyLinearVelocity:Lerp(desiredVel, 0.6)

            local toTarget = targetPos - myPos
            if toTarget.Magnitude > 0.5 then
                local goalCF = CFrame.lookAt(myPos, targetPos)
                local diffCF = root.CFrame:Inverse() * goalCF
                local rx, ry, rz = diffCF:ToEulerAnglesXYZ()
                rx = math.clamp(rx, -1.5, 1.5)
                ry = math.clamp(ry, -2.0, 2.0)
                rz = math.clamp(rz, -1.5, 1.5)
                root.AssemblyAngularVelocity = root.CFrame:VectorToWorldSpace(Vector3.new(rx * 30, ry * 30, rz * 30))
            end

            if targetDist and targetDist <= _v2HIT_DIST then
                _v2TrySwing()
            end
        end)
    end

        _v2HitCD = false
        local c = LP.Character

    local function updateProgressBar(p)

    local function isMyPlotByName(plotName)
        local plots = workspace:FindFirstChild("Plots")
        if not plots then return false end
        local plot = plots:FindFirstChild(plotName)
        if not plot then return false end
        local sign = plot:FindFirstChild("PlotSign")
        if sign then
            local yourBase = sign:FindFirstChild("YourBase")
            if yourBase and yourBase:IsA("BillboardGui") then
                return yourBase.Enabled == true
            end
        end
        return false
    end

    local function findNearestPrompt()
        local hrp = getHRP()
        if not hrp then return nil end
        local plots = workspace:FindFirstChild("Plots")
        if not plots then return nil end
        local nearest, dist = nil, math.huge
        for _, plot in ipairs(plots:GetChildren()) do
            if isMyPlotByName(plot.Name) then continue end
            local pods = plot:FindFirstChild("AnimalPodiums")
            if not pods then continue end
            for _, pod in ipairs(pods:GetChildren()) do
                local base = pod:FindFirstChild("Base")
                if not base then continue end
                local spawn = base:FindFirstChild("Spawn")
                if not spawn then continue end
                local d = (spawn.Position - hrp.Position).Magnitude
                if d <= Steal.StealRadius and d < dist then
                    local att = spawn:FindFirstChild("PromptAttachment")
                    if att then
                        for _, p in ipairs(att:GetChildren()) do
                            if p:IsA("ProximityPrompt") then
                                if p.ActionText and p.ActionText:find("Steal") then
                                    nearest, dist = p, d
                                end
                            end
                        end
                    end
                end
            end
        end
        return nearest
    end

    local function executeSteal(prompt)
        if isStealing then return end

        if not StealDataCache[prompt] then
            StealDataCache[prompt] = {hold = {}, trigger = {}, ready = true}
            if getconnections then
                for _, c in ipairs(getconnections(prompt.PromptButtonHoldBegan)) do
                    if c.Function then
                        table.insert(StealDataCache[prompt].hold, c.Function)
                    end
                end
                for _, c in ipairs(getconnections(prompt.Triggered)) do
                    if c.Function then
                        table.insert(StealDataCache[prompt].trigger, c.Function)
                    end
                end
            end
        end

        local data = StealDataCache[prompt]
        if not data.ready then return end

        data.ready = false
        isStealing = true
        State.isStealing = true
        local startTime = tick()

        task.spawn(function()
            for _, f in ipairs(data.hold) do pcall(f) end
            while tick() - startTime < Steal.StealDuration do
                local elapsed = tick() - startTime
                updateProgressBar(math.clamp(elapsed / Steal.StealDuration, 0, 1))
                task.wait()
            end
            updateProgressBar(1)
            for _, f in ipairs(data.trigger) do pcall(f) end
            task.wait(0.05)
            resetProgressBar()
            data.ready = true
            isStealing = false
            State.isStealing = false
        end)
    end

        Conns.autoSteal = RunService.Heartbeat:Connect(function()
            if not Steal.AutoStealEnabled or isStealing then return end
            local ok, prompt = pcall(findNearestPrompt)

        end)

        end

    end

        _dropConn = RunService.Heartbeat:Connect(function()
            local c = LP.Character

                if _dropConn then _dropConn:Disconnect(); _dropConn = nil end
                State.dropEnabled = false
                if stackBtnRefs.drop then stackBtnRefs.drop.setOn(false) end
                return
            end
 
            if not State.dropEnabled then
                if _dropConn then _dropConn:Disconnect(); _dropConn = nil end
                if stackBtnRefs.drop then stackBtnRefs.drop.setOn(false) end
                return
            end
 
            if tick() - t0 < DROP_ASCEND_DURATION then
                local lv = r.AssemblyLinearVelocity
                r.AssemblyLinearVelocity = Vector3.new(lv.X, DROP_ASCEND_SPEED, lv.Z)
                return
            end
            
            if _dropConn then _dropConn:Disconnect(); _dropConn = nil end
            pcall(function()
                local rp = RaycastParams.new()
                rp.FilterDescendantsInstances = {c}
                rp.FilterType = Enum.RaycastFilterType.Exclude
                local rr = workspace:Raycast(r.Position, Vector3.new(0, -3000, 0), rp)
                if rr then
                    local hum = c:FindFirstChildOfClass("Humanoid")
                    local offset = ((hum and hum.HipHeight) or 2) + (r.Size.Y / 2)
                    r.CFrame = CFrame.new(r.Position.X, rr.Position.Y + offset, r.Position.Z)
                    r.AssemblyLinearVelocity = Vector3.new(0, -15, 0)
                else
                    r.CFrame = CFrame.new(r.Position.X, -7, r.Position.Z)
                    r.AssemblyLinearVelocity = Vector3.zero
                end
            end)
            State.dropEnabled = false
            if stackBtnRefs.drop then stackBtnRefs.drop.setOn(false) end
        end)
    end

        State.dropEnabled = false
        if _dropConn then _dropConn:Disconnect(); _dropConn = nil end

    end

    -- ============================================================
    local noclipParts = {}

    local function isBaseAccessory(part)
        if not part or not part:IsA("BasePart") then return false end
        local parent = part.Parent
        while parent and parent ~= workspace do
            if parent:IsA("Model") and (parent.Name:lower():find("base") or parent.Name:lower():find("plot")) then
                if part.Size.Magnitude < 20 and part.Transparency > 0 then
                    return true
                end
                local name = part.Name:lower()
                if name:find("pumpkin") or name:find("decoration") or name:find("accessory") or 
                   name:find("summer") or name:find("flower") or name:find("plant") or
                   name:find("rock") or name:find("tree") or name:find("bush") then
                    return true
                end
            end
            parent = parent.Parent
        end
        return false
    end

    local function handleNoclip(part)
        if not part:IsA("BasePart") then return end
        if isBaseAccessory(part) then
            if not noclipParts[part] then
                noclipParts[part] = part.CanCollide
            end
            part.CanCollide = false
        end
    end

    startNoclip = function()
        if Conns.noclip then return end
        noclipParts = {}
        for _, part in ipairs(workspace:GetDescendants()) do
            handleNoclip(part)
        end
        Conns.noclip = workspace.DescendantAdded:Connect(handleNoclip)
    end

    stopNoclip = function()
        if Conns.noclip then
            Conns.noclip:Disconnect()
            Conns.noclip = nil
        end
        for part, oldCollide in pairs(noclipParts) do
            pcall(function()
                if part and part.Parent then
                    part.CanCollide = oldCollide
                end
            end)
        end
        noclipParts = {}
    end

    setNoclip = function(on)
        State.noclipEnabled = on
        if on then
            startNoclip()
        else
            stopNoclip()
        end
        pcall(saveConfig)
    end

    -- ============================================================
    -- RAGDOLL TIMER
    -- ============================================================
    local ragdollTimerActive = false
    local ragdollTimerToken = 0
    local ragdollLabel = nil

    local function setupRagdollTimer(char)
        if not char then return end
        local head = char:FindFirstChild("Head")
        if not head then return end
        local old = head:FindFirstChild("RagdollTimerBB")
        if old then old:Destroy() end

        local bb = Instance.new("BillboardGui")
        bb.Name = "RagdollTimerBB"
        bb.Size = UDim2.new(0, 180, 0, 60)
        bb.StudsOffset = Vector3.new(0, 5, 0)
        bb.AlwaysOnTop = true
        bb.Enabled = true
        bb.Parent = head

        local lbl = Instance.new("TextLabel")
        lbl.Name = "Countdown"
        lbl.Size = UDim2.new(1, 0, 1, 0)
        lbl.BackgroundTransparency = 1
        lbl.Text = ""
        lbl.Visible = false
        lbl.TextScaled = true
        lbl.Font = Enum.Font.GothamBold
        lbl.TextColor3 = Color3.fromRGB(255, 0, 0)
        lbl.TextStrokeTransparency = 0
        lbl.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        lbl.Parent = bb

        ragdollLabel = lbl
    end

    local function startRagdollTimer()
        if ragdollTimerActive then return end
        if not ragdollLabel or not ragdollLabel.Parent then
            setupRagdollTimer(LP.Character)
        end
        if not ragdollLabel then return end

        ragdollTimerActive = true
        ragdollTimerToken = ragdollTimerToken + 1
        local token = ragdollTimerToken
        local lbl = ragdollLabel

        task.spawn(function()
            lbl.Visible = true
            for i = 3, 1, -1 do
                if token ~= ragdollTimerToken then return end
                if not State.ragdollTimerEnabled then
                    lbl.Visible = false
                    lbl.Text = ""
                    ragdollTimerActive = false
                    return
                end
                lbl.Text = tostring(i)
                task.wait(1)
            end
            if token ~= ragdollTimerToken then return end
            if not State.ragdollTimerEnabled then
                lbl.Visible = false
                lbl.Text = ""
                ragdollTimerActive = false
                return
            end
            lbl.Text = "GO!"
            repeat
                task.wait(0.1)
                if not State.ragdollTimerEnabled then
                    lbl.Visible = false
                    lbl.Text = ""
                    ragdollTimerActive = false
                    return
                end
                local char = LP.Character
                local hum = char and char:FindFirstChildOfClass("Humanoid")
                if not hum then break end
                local state = hum:GetState()
                if state ~= Enum.HumanoidStateType.Physics
                and state ~= Enum.HumanoidStateType.Ragdoll
                and state ~= Enum.HumanoidStateType.FallingDown then
                    break
                end
            until false
            if token ~= ragdollTimerToken then return end
            if not State.ragdollTimerEnabled then
                lbl.Visible = false
                lbl.Text = ""
                ragdollTimerActive = false
                return
            end
            task.wait(0.25)
            lbl.Visible = false
            lbl.Text = ""
            ragdollTimerActive = false
        end)
    end

    RunService.Heartbeat:Connect(function()
        if not State.ragdollTimerEnabled then return end
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        local state = hum:GetState()
        if state == Enum.HumanoidStateType.Physics
        or state == Enum.HumanoidStateType.Ragdoll
        or state == Enum.HumanoidStateType.FallingDown then
            startRagdollTimer()
        end
    end)

    end)

    -- ============================================================
    local function fixCamera()
        pcall(function()
            local cam = Workspace.CurrentCamera
            if cam then
                cam.CameraType = Enum.CameraType.Custom
                cam.FieldOfView = 70
            end
        end)
    end

    local function startCameraFix()
        if Conns.cameraFix then return end
        Conns.cameraFix = RunService.RenderStepped:Connect(function()
            pcall(function()
                local cam = Workspace.CurrentCamera
                if cam then
                    if cam.CameraType ~= Enum.CameraType.Custom then
                        cam.CameraType = Enum.CameraType.Custom
                    end
                    if State.stretchedResEnabled then
                        if cam.FieldOfView ~= State.stretchFOV then
                            cam.FieldOfView = State.stretchFOV
                        end
                    else
                        if cam.FieldOfView ~= 70 then
                            cam.FieldOfView = 70
                        end
                    end
                end
            end)
        end)
    end

    -- ============================================================
    -- LOAD CONFIG
    -- ============================================================
    loadConfig = function()
        local data = _readfile(CONFIG_FILE)
        if not data then 
            print("[Bont Hub UI No config file found")
            return 
        end
        
        local ok, cfg = pcall(function()
            return HttpService:JSONDecode(data)
        end)
        
        if not ok or not cfg then
            print("[Bont Hub UI] Failed to parse config")
            return 
        end

        local function safeSetText(element, text)
            if element and element:IsA("TextBox") then
                pcall(function()
                    element.Text = tostring(text)
                end)
            end
        end

        local function safeSetToggle(element, on)
            if element and type(element) == "function" then
                pcall(function() element(on) end)
            end
        end

        if cfg.normalSpeed then
            State.normalSpeed = cfg.normalSpeed
            safeSetText(normalBox, cfg.normalSpeed)
        end
        
        if cfg.carrySpeed then
            State.carrySpeed = cfg.carrySpeed
            safeSetText(carryBox, cfg.carrySpeed)
        end
        
        if cfg.laggerSpeed then
            State.laggerSpeed = cfg.laggerSpeed
            safeSetText(laggerBox, cfg.laggerSpeed)
        end
        
        if cfg.uiScale then
            State.uiScale = cfg.uiScale
            local mainScale = mainOuter and mainOuter:FindFirstChild("MainScale")
            if mainScale then
                pcall(function() mainScale.Scale = cfg.uiScale end)
            end
            safeSetText(uiScaleBox, cfg.uiScale)
        end
        
        if cfg.stealBarScale then
            State.stealBarScale = cfg.stealBarScale
            safeSetText(stealBarScaleBox, cfg.stealBarScale)
            if stealBarOuter then
                pcall(function()
                    stealBarOuter.Size = UDim2.new(0, 340 * cfg.stealBarScale, 0, 42 * cfg.stealBarScale)
                    stealBarOuter.Position = UDim2.new(0.5, -170 * cfg.stealBarScale, 0, 8)
                end)
            end
        end
        
        if cfg.stretchedResEnabled ~= nil then
            State.stretchedResEnabled = cfg.stretchedResEnabled
            if State.stretchedResEnabled then 
                pcall(enableStretchRez) 
            end
        end
        
        if cfg.stretchFOV then
            State.stretchFOV = cfg.stretchFOV
        end
        
        if cfg.noclipEnabled ~= nil then
            State.noclipEnabled = cfg.noclipEnabled
            if State.noclipEnabled then 
                pcall(startNoclip) 
            end
        end
        
        if cfg.ragdollTimerEnabled ~= nil then
            State.ragdollTimerEnabled = cfg.ragdollTimerEnabled
        end
        
        if cfg.flingProtection ~= nil then
            State.flingProtection = cfg.flingProtection
            if State.flingProtection then 
                pcall(startFlingProtection) 
            end
        end

        if cfg.speedKey then
            pcall(function() Keys.speed = Enum.KeyCode[cfg.speedKey] end)
            if keybindBtnRefs.speed then
                pcall(function() keybindBtnRefs.speed.Text = cfg.speedKey end)
            end
        end
        
        if cfg.laggerKey then
            pcall(function() Keys.lagger = Enum.KeyCode[cfg.laggerKey] end)
            if keybindBtnRefs.lagger then
                pcall(function() keybindBtnRefs.lagger.Text = cfg.laggerKey end)
            end
        end
        
        if cfg.autoLeftKey then
            pcall(function() Keys.autoLeft = Enum.KeyCode[cfg.autoLeftKey] end)
            if keybindBtnRefs.autoLeft then
                pcall(function() keybindBtnRefs.autoLeft.Text = cfg.autoLeftKey end)
            end
        end
        
        if cfg.autoRightKey then
            pcall(function() Keys.autoRight = Enum.KeyCode[cfg.autoRightKey] end)
            if keybindBtnRefs.autoRight then
                pcall(function() keybindBtnRefs.autoRight.Text = cfg.autoRightKey end)
            end
        end
        
        if cfg.aimbotKey then
            pcall(function() Keys.aimbot = Enum.KeyCode[cfg.aimbotKey] end)
            if keybindBtnRefs.aimbot then
                pcall(function() keybindBtnRefs.aimbot.Text = cfg.aimbotKey end)
            end
        end
        
        if cfg.batV2Key then
            pcall(function() Keys.batV2 = Enum.KeyCode[cfg.batV2Key] end)
            if keybindBtnRefs.batV2 then
                pcall(function() keybindBtnRefs.batV2.Text = cfg.batV2Key end)
            end
        end

            pcall(function() Keys.drop = Enum.KeyCode[cfg.dropKey] end)
            if keybindBtnRefs.drop then
                pcall(function() keybindBtnRefs.drop.Text = cfg.dropKey end)
            end

        if cfg.tpDownKey then
            pcall(function() Keys.tpDown = Enum.KeyCode[cfg.tpDownKey] end)
            if keybindBtnRefs.tpDown then
                pcall(function() keybindBtnRefs.tpDown.Text = cfg.tpDownKey end)
            end

            pcall(function() Keys.guiHide = Enum.KeyCode[cfg.guiHideKey] end)
            if keybindBtnRefs.guiHide then

        end

        end
        
        if cfg.medusaCounter then
            State.medusaCounterEnabled = true
            pcall(setupMedusaCounter, LP.Character)

        end
        
        if cfg.batCounter then
            State.batCounterEnabled = true

            safeSetToggle(setBatCounter, true)
        end
        
        if cfg.unwalk then

            pcall(startUnwalk)

        if cfg.stackPositions then

                for i, def in ipairs(stackDefs) do
                    local posData = cfg.stackPositions[def.key]

                        if wrapper then
                            pcall(function()
                                wrapper.Position = UDim2.new(
                                    posData.X.Scale, posData.X.Offset,
                                    posData.Y.Scale, posData.Y.Offset
                                )
                            end)
                        end
                    end
                end
            end)
        end

        if cfg.stealBarPos then
            task.defer(function()
                if stealBarOuter then
                    pcall(function()
                        stealBarOuter.Position = UDim2.new(
                            cfg.stealBarPos.X.Scale, cfg.stealBarPos.X.Offset,
                            cfg.stealBarPos.Y.Scale, cfg.stealBarPos.Y.Offset
                        )
                    end)
                end
            end)
        end
        
        pcall(fixCamera)
        pcall(startCameraFix)
        
        print("[Bont Hub UI] Config loaded successfully!")
    end

    task.spawn(function()
        task.wait(0.5)
        pcall(loadConfig)
    end)
